// One-off local measurement (not repo code) of the single verification_state row lock (S-12 / E.2).
// Runs against a disposable local PostgreSQL in a random schema; never against Supabase.
import { randomUUID } from 'node:crypto';
import { createRequire } from 'node:module';
import { resolve } from 'node:path';
import { pathToFileURL } from 'node:url';

const WEB = 'C:/Users/Maulana/Pictures/Matkul IF/Semester 7/Blockchain/UAS/Degree-E-Sign-and-Verifier-ZamaFHE-main/apps/web';
const fromWeb = createRequire(resolve(WEB, 'package.json'));
const url = process.env.BENCH_DATABASE_URL!;
if (!['127.0.0.1', 'localhost'].includes(new URL(url).hostname)) throw new Error('local database only');

async function main() {
  const { Pool } = fromWeb('pg');
  const { migrate } = await import(pathToFileURL(fromWeb.resolve('drizzle-orm/node-postgres/migrator')).href);
  const { createDatabase } = await import(pathToFileURL(resolve(WEB, 'src/server/db/client.ts')).href);
  const { withDatabaseState } = await import(pathToFileURL(resolve(WEB, 'src/server/db/state.ts')).href);
  const schema = `bench_${randomUUID().replaceAll('-', '')}`;
  const admin = new Pool({ connectionString: url, max: 1 });
  await admin.query(`CREATE SCHEMA "${schema}"`);
  const scoped = new URL(url); scoped.searchParams.set('options', `-c search_path=${schema}`);
  const setup = createDatabase(scoped.toString(), 1);
  await migrate(setup.db, { migrationsFolder: resolve(WEB, 'drizzle'), migrationsSchema: schema });
  await withDatabaseState(setup.db, (state: any) => {
    for (let i = 0; i < 500; i++) state.sessions[`s${i}`] = { id: `s${i}`, csrf: 'x'.repeat(64), expiresAt: '2099-01-01T00:00:00.000Z' };
    for (let i = 0; i < 200; i++) state.jobs[`j${i}`] = { id: `j${i}`, owner: `s${i}`, idempotencyKey: `k${i}`, status: 'COMPLETED', fileName: 'contoh.pdf', fileSize: 1000,
      mimeType: 'application/pdf', createdAt: '', expiresAt: '2099-01-01T00:00:00.000Z', artifactsExpireAt: '2099-01-01T00:00:00.000Z', mode: 'testnet', synthetic: false, attempts: 0 };
    state.rates.bench = { count: 0, resetAt: 1 };
  });
  const size = (await admin.query(`SELECT pg_column_size(body) AS bytes FROM "${schema}".verification_state WHERE id = 1`)).rows[0].bytes;
  console.log(JSON.stringify({ environment: 'local Docker postgres:16-alpine (127.0.0.1), not Supabase', stateRowBytes: size, sessions: 500, jobs: 200, operationsPerLevel: 200 }));
  for (const concurrency of [1, 5, 20, 50]) {
    const pool = createDatabase(scoped.toString(), Math.min(concurrency, 10));
    const latencies: number[] = []; let errors = 0; let next = 0;
    const started = performance.now();
    await Promise.all(Array.from({ length: concurrency }, async () => {
      while (next < 200) {
        next++;
        const t0 = performance.now();
        try { await withDatabaseState(pool.db, (state: any) => { state.rates.bench.count++; }); latencies.push(performance.now() - t0); }
        catch { errors++; }
      }
    }));
    const seconds = (performance.now() - started) / 1000;
    latencies.sort((a, b) => a - b);
    const q = (p: number) => latencies[Math.min(latencies.length - 1, Math.floor(p * latencies.length))]!.toFixed(1);
    console.log(JSON.stringify({ concurrency, poolMax: Math.min(concurrency, 10), completed: latencies.length, errors, seconds: seconds.toFixed(2),
      throughputOpsPerSecond: (latencies.length / seconds).toFixed(1), p50ms: q(0.5), p95ms: q(0.95), maxMs: latencies.at(-1)!.toFixed(1) }));
    await pool.pool.end();
  }
  const final = await withDatabaseState(setup.db, (state: any) => state.rates.bench.count);
  console.log(JSON.stringify({ counterAfterAllLevels: final, expected: 800, lostUpdates: 800 - final }));
  await setup.pool.end();
  await admin.query(`DROP SCHEMA "${schema}" CASCADE`);
  await admin.end();
}
main().catch(error => { console.error('bench error:', error.message); process.exitCode = 1; });
